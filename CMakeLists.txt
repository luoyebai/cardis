cmake_minimum_required(VERSION 3.24)
project(cardis VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CARDIS_BUILD_CLIENT "Build the desktop client" ON)
include(CTest)
include(cmake/Dependencies.cmake)

function(cardis_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
    endif()
endfunction()

add_library(cardis_core src/core/game.cpp src/core/character.cpp)
target_include_directories(cardis_core PUBLIC include)
cardis_warnings(cardis_core)
add_library(cardis_content src/content/catalog.cpp src/content/roster.cpp)
target_include_directories(cardis_content PUBLIC include)
target_link_libraries(cardis_content PUBLIC cardis_core PRIVATE nlohmann_json::nlohmann_json)
cardis_warnings(cardis_content)
add_library(cardis_runtime src/runtime/runtime.cpp)
target_include_directories(cardis_runtime PUBLIC include)
target_link_libraries(cardis_runtime PUBLIC cardis_content cordis_core)
cardis_warnings(cardis_runtime)

add_executable(cardis_cli apps/cli/main.cpp)
target_link_libraries(cardis_cli PRIVATE cardis_runtime)
cardis_warnings(cardis_cli)
if(CARDIS_BUILD_CLIENT)
    add_library(cardis_gui STATIC apps/client/raygui.cpp)
    target_include_directories(cardis_gui SYSTEM PUBLIC ${raygui_SOURCE_DIR}/src)
    target_link_libraries(cardis_gui PUBLIC raylib)
    add_executable(cardis apps/client/main.cpp)
    target_link_libraries(cardis PRIVATE cardis_runtime cardis_gui)
    cardis_warnings(cardis)
    add_custom_target(cardis_assets ALL
        COMMAND ${CMAKE_COMMAND} -E copy_directory ${PROJECT_SOURCE_DIR}/assets $<TARGET_FILE_DIR:cardis>/assets
        DEPENDS cardis
        COMMENT "Syncing Cardis character and UI assets")
endif()
if(BUILD_TESTING)
    # Compile each public header in isolation to enforce self-contained includes.
    foreach(header core/game core/character content/catalog runtime/runtime)
        string(REPLACE "/" "_" header_id "${header}")
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${header_id}_header.cpp" "#include <cardis/${header}.hpp>\n")
        add_library(cardis_header_${header_id} OBJECT "${CMAKE_CURRENT_BINARY_DIR}/${header_id}_header.cpp")
        target_link_libraries(cardis_header_${header_id} PRIVATE cardis_runtime)
        cardis_warnings(cardis_header_${header_id})
    endforeach()
    add_executable(cardis_tests tests/game_test.cpp)
    target_link_libraries(cardis_tests PRIVATE cardis_runtime)
    cardis_warnings(cardis_tests)
    add_test(NAME cardis_rules COMMAND cardis_tests ${PROJECT_SOURCE_DIR}/assets/cards.json)
    add_test(NAME cardis_cli_smoke COMMAND cardis_cli ${PROJECT_SOURCE_DIR}/assets/cards.json)
    add_executable(cardis_character_tests tests/character_test.cpp)
    target_link_libraries(cardis_character_tests PRIVATE cardis_runtime nlohmann_json::nlohmann_json)
    cardis_warnings(cardis_character_tests)
    add_test(NAME cardis_characters COMMAND cardis_character_tests ${PROJECT_SOURCE_DIR}/assets/cards.json)
    set_tests_properties(cardis_rules cardis_cli_smoke cardis_characters PROPERTIES TIMEOUT 60)
endif()
